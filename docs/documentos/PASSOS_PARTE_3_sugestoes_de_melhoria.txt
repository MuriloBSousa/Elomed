Passo 1.2 - Validar CPF e data de nascimento no Validator
Adicione estes dois metodos no final da classe Validator
(app/helpers/Validator.php). O primeiro valida o digito verificador do
CPF; o segundo impede data futura e datas antes de 1900 (vai servir tb
para a sugestao 6.1):

    public function cpf(string $field, string $message = null): self {
        if (isset($this->data[$field]) && !empty($this->data[$field])) {
            $cpf = preg_replace('/\D/', '', $this->data[$field]);
            if (strlen($cpf) !== 11) {
                $this->errors[$field] = $message ?? "O campo {$field} deve ter 11 digitos";
                return $this;
            }
            if (preg_match('/^(\d)\1{10}$/', $cpf)) {
                $this->errors[$field] = $message ?? "O campo {$field} e invalido";
                return $this;
            }
            for ($t = 9; $t < 11; $t++) {
                $soma = 0;
                for ($i = 0; $i < $t; $i++) {
                    $soma += (int) $cpf[$i] * (($t + 1) - $i);
                }
                $digito = (($soma * 10) % 11) % 10;
                if ((int) $cpf[$t] !== $digito) {
                    $this->errors[$field] = $message ?? "O campo {$field} e invalido";
                    return $this;
                }
            }
        }
        return $this;
    }

    public function dateNascimento(string $field, string $message = null): self {
        if (isset($this->data[$field]) && !empty($this->data[$field])) {
            $ts = strtotime($this->data[$field]);
            if (!$ts || $this->data[$field] !== date('Y-m-d', $ts)) {
                $this->errors[$field] = $message ?? "O campo {$field} deve ser uma data valida";
                return $this;
            }
            if ($ts > strtotime('today')) {
                $this->errors[$field] = $message ?? "A {$field} nao pode estar no futuro";
                return $this;
            }
            if ((int) date('Y', $ts) < 1900) {
                $this->errors[$field] = $message ?? "A {$field} e muito antiga (antes de 1900)";
                return $this;
            }
        }
        return $this;
    }

(Opcional) Funcao que formata CPF e telefone para mostrar bonito na tela:

    function formatarCpf($cpf) {
        $d = preg_replace('/\D/', '', $cpf);
        if (strlen($d) !== 11) return $cpf;
        return preg_replace('/(\d{3})(\d{3})(\d{3})(\d{2})/', '$1.$2.$3-$4', $d);
    }

    function formatarTelefone($tel) {
        $d = preg_replace('/\D/', '', $tel);
        if (strlen($d) === 11) return '(' . substr($d, 0, 2) . ') ' . substr($d, 2, 5) . '-' . substr($d, 7);
        if (strlen($d) === 10) return '(' . substr($d, 0, 2) . ') ' . substr($d, 2, 4) . '-' . substr($d, 6);
        return $tel;
    }

    public function buscarPorNomeOuCpfPaciente($termo) {
        $t = '%' . $termo . '%';
        $stmt = $this->db->prepare(
            "SELECT u.id, u.nome, u.email, u.perfil, p.cpf
               FROM tbl_usuario u
          LEFT JOIN tbl_paciente p ON p.fk_usuario = u.id
              WHERE u.perfil = 'paciente'
                AND (u.nome LIKE :termo OR p.cpf LIKE :termo)

    public function getLinhaDoTempo($pacienteId) {
        $sql = "SELECT 'agendamento' AS tipo,
                       DATE_FORMAT(a.data_agendamento, '%d/%m/%Y') AS data_evento,
                       a.horario_inicio AS horario, a.status AS status,
                       a.justificativa_cancelamento AS detalhe
                  FROM tbl_agendamento a WHERE a.fk_paciente = :id
                 UNION ALL
                SELECT 'triagem',
                       DATE_FORMAT(t.criado_em, '%d/%m/%Y'),
                       NULL, t.status,
                       CONCAT('Solicitante: ', t.nome_solicitante,
                              IF(t.classificacao IS NULL, '', CONCAT(' | Risco: ', t.classificacao)))
                  FROM tbl_triagem t WHERE t.fk_paciente = :id
                 UNION ALL
                SELECT 'visita',
                       DATE_FORMAT(v.horario_inicio, '%d/%m/%Y'),
                       TIME(v.horario_inicio), v.status, NULL
                  FROM tbl_visita v WHERE v.fk_paciente = :id
                 ORDER BY STR_TO_DATE(data_evento, '%d/%m/%Y') DESC, horario DESC";
        $stmt = $this->db->prepare($sql);
        $stmt->execute(['id' => $pacienteId]);
        return $stmt->fetchAll(PDO::FETCH_ASSOC);
    }

    public function verificarConflito($fkProfissional, $data, $inicio, $fim) {
        $sql = "SELECT id, horario_inicio, horario_fim
                  FROM tbl_agendamento
                 WHERE fk_profissional = :prof AND data_agendamento = :data
                   AND status IN ('pendente', 'agendado', 'em_andamento')";
        $stmt = $this->db->prepare($sql);
        $stmt->execute(['prof' => $fkProfissional, 'data' => $data]);
        foreach ($stmt->fetchAll() as $e) {
            if ($inicio < $e['horario_fim'] && $e['horario_inicio'] < $fim) {
                return $e;
            }
        }
        return null;
    }

    <?php
    require_once __DIR__ . '/Model.php';

    class Auditoria extends Model {
        protected $table = 'tbl_auditoria';

        public function registrar(string $acao, string $descricao, array $propriedades = []) {
            $stmt = $this->db->prepare(
                "INSERT INTO tbl_auditoria
                    (fk_usuario, usuario_nome, usuario_perfil, acao, descricao, propriedades, ip)
                 VALUES
                    (:fk, :nome, :perfil, :acao, :descricao, :props, :ip)"
            );
            return $stmt->execute([
                'fk'         => Session::get('usuario_id'),
                'nome'       => Session::get('usuario_nome'),
                'perfil'     => Session::get('usuario_perfil'),
                'acao'       => $acao,
                'descricao'  => $descricao,
                'props'      => json_encode($propriedades),
                'ip'         => $_SERVER['REMOTE_ADDR'] ?? 'unknown'
            ]);
        }

        public function listar(string $where, array $params, int $limite, int $offset) {
            $sql = "SELECT id, acao, descricao, usuario_nome, usuario_perfil, ip, data_hora, propriedades
                      FROM tbl_auditoria{$where}
                     ORDER BY data_hora DESC
                     LIMIT :limite OFFSET :offset";
            $stmt = $this->db->prepare($sql);
            foreach ($params as $k => $v) {
                $stmt->bindValue(':' . $k, $v);
            }
            $stmt->bindValue(':limite', $limite, PDO::PARAM_INT);
            $stmt->bindValue(':offset', $offset, PDO::PARAM_INT);
            $stmt->execute();
            return $stmt->fetchAll(PDO::FETCH_ASSOC);
        }

        public function contar(string $where, array $params) {
            $stmt = $this->db->prepare("SELECT COUNT(*) FROM tbl_auditoria{$where}");
            $stmt->execute($params);
            return (int) $stmt->fetchColumn();
        }
    }

    class AuditoriaController {
        private $auditoriaModel;

        public function __construct() {
            Session::start();
            if (!Session::has('usuario_id')) {
                header('Location: index.php?controller=login&action=index');
                exit;
            }
            if (Session::get('usuario_perfil') !== Constants::PERFIL_ADMIN) {
                header('Location: index.php?controller=painel&action=index');
                exit;
            }
            $this->auditoriaModel = new Auditoria();
        }

        public function index() {
            $acao = trim($_GET['acao'] ?? '');
            $por  = trim($_GET['por']  ?? '');
            $de   = trim($_GET['data_inicio'] ?? '');
            $ate  = trim($_GET['data_fim'] ?? '');
            $pagina = max(1, (int) ($_GET['page'] ?? 1));
            $porPagina = 15;

            $onde   = [];
            $params = [];

            if ($acao !== '') {
                $onde[] = 'acao = :acao';
                $params['acao'] = $acao;
            }
            if ($por !== '') {
                $onde[] = '(usuario_nome LIKE :por OR usuario_perfil LIKE :por)';
                $params['por'] = '%' . $por . '%';
            }
            if ($de !== '') {
                $onde[] = 'DATE(data_hora) >= :de';
                $params['de'] = $de;
            }
            if ($ate !== '') {
                $onde[] = 'DATE(data_hora) <= :ate';
                $params['ate'] = $ate;
            }
            $where = $onde ? ' WHERE ' . implode(' AND ', $onde) : '';

            $filtros = array_filter([
                'controller'  => 'auditoria',
                'action'      => 'index',
                'acao'        => $acao,
                'por'         => $por,
                'data_inicio' => $de,
                'data_fim'    => $ate
            ], function ($v) { return $v !== ''; });

            $total = $this->auditoriaModel->contar($where, $params);
            $pagination = new Pagination($total, $porPagina, $pagina, 'index.php', $filtros);

            $registros = $this->auditoriaModel->listar(
                $where, $params, $pagination->getLimit(), $pagination->getOffset()
            );

            $acoesDisponiveis = $this->auditoriaModel->listarAcoes();

            require_once __DIR__ . '/../views/auditoria/index.php';
        }

        public function pdf() {
            $acao = trim($_GET['acao'] ?? '');
            $por  = trim($_GET['por'] ?? '');
            $de   = trim($_GET['data_inicio'] ?? '');
            $ate  = trim($_GET['data_fim'] ?? '');

            $onde   = [];
            $params = [];
            if ($acao !== '') { $onde[] = 'acao = :acao';            $params['acao'] = $acao; }
            if ($por  !== '') { $onde[] = '(usuario_nome LIKE :por OR usuario_perfil LIKE :por)'; $params['por'] = '%' . $por . '%'; }
            if ($de   !== '') { $onde[] = 'DATE(data_hora) >= :de';  $params['de'] = $de; }
            if ($ate  !== '') { $onde[] = 'DATE(data_hora) <= :ate'; $params['ate'] = $ate; }
            $where = $onde ? ' WHERE ' . implode(' AND ', $onde) : '';

            $registros = $this->auditoriaModel->listar($where, $params, 10000, 0);

            require_once dirname(__DIR__, 2) . '/libs/autoload.php';

            ob_start();
            require __DIR__ . '/../views/auditoria/pdf.php';
            $html = ob_get_clean();

            $dompdf = new Dompdf\Dompdf();
            $dompdf->loadHtml($html, 'UTF-8');
            $dompdf->setPaper('A4', 'landscape');
            $dompdf->render();
            $dompdf->stream('trilha_auditoria_' . date('Ymd_His') . '.pdf', ['Attachment' => false]);
            exit;
        }
    }
